# 🎤 AI Interview Tool (Google Apps Script)

## 🚀 USER GUIDE (Start Here)

### 1. Setup

1. Open the Google Sheet template
2. Go to menu: **🎤 Strat Interview**
3. Click **▶ New interview**

---

### 2. Run an Interview

* A wizard opens (7 sections)
* Fill step-by-step:

  * S1 → Context
  * S2 → AI Projects
  * S3 → Market impact
  * S4 → Processes
  * S5 → Opportunities
  * S6 → Execution
  * S7 → Wrap-up

✅ Data is **auto-saved in background**

## 🧠 MACRO LOGIC (How it works)

### 1. Data Model = Multi-table relational system

Each interview is identified by:

```
interview_id
```

And stored across multiple tables:

| Table           | Purpose            |
| --------------- | ------------------ |
| DB_INTERVIEW    | Metadata           |
| DB_SCOPE_VISION | Core + notes       |
| DB_PROJECTS     | AI projects        |
| DB_MARKET       | Porter forces      |
| DB_PROCESSES    | Process analysis   |
| DB_EXECUTION    | Execution maturity |
| REPORTS         | Aggregated outputs |

---

### 2. Wizard → Structured Data Pipeline

UI (Wizard) → JSON payload → Mapping → Sheets

```
User Input (Wizard UI)
        ↓
Structured JSON (per section)
        ↓
Validation (allowed keys)
        ↓
Mapping to DB fields
        ↓
Batch write to Sheets
```

---

### 3. Section-based architecture

Each section:

* Has **allowed input keys**
* Maps to:

  * Scalar fields (1 row)
  * or child tables (multiple rows)

Example:

* S2 → Projects → multiple rows
* S3 → Market → fixed 4 forces
* S4 → Processes → dynamic rows

---

### 4. Performance design

Optimized for Apps Script limits:

* ✅ Read once → process in memory → write once
* ✅ Batch updates
* ✅ Locking for concurrent writes
* ✅ Lazy loading in UI

---

### 5. Save strategy

* Auto-save every ~8 seconds
* Background sync
* Final full save on completion

function isPlainObject(value) {
  return Object.prototype.toString.call(value) === '[object Object]';
}

function validateArrayOfObjects(sectionNum, fieldName, value) {
  if (Object.prototype.toString.call(value) !== '[object Array]') {
    throw new Error(
      'validateSectionPayload: "' + fieldName + '" for section "' + sectionNum + '" must be an array.'
    );
  }

  for (var i = 0; i < value.length; i++) {
    if (!isPlainObject(value[i])) {
      throw new Error(
        'validateSectionPayload: "' + fieldName + '" item #' + (i + 1) +
        ' for section "' + sectionNum + '" must be an object.'
      );
    }
  }
}

function validateSectionPayload(sectionNum, data) {
  var config = SECTION_FIELD_MAP[sectionNum];
  if (!config) {
    throw new Error('validateSectionPayload: unknown section "' + sectionNum + '".');
  }

  if (!isPlainObject(data)) {
    throw new Error('validateSectionPayload: payload for section "' + sectionNum + '" must be an object.');
  }

  var allowed = config.allowedKeys || [];
  var incomingKeys = Object.keys(data);
  var unknownKeys = [];

  for (var i = 0; i < incomingKeys.length; i++) {
    if (allowed.indexOf(incomingKeys[i]) === -1) {
      unknownKeys.push(incomingKeys[i]);
    }
  }

  if (unknownKeys.length) {
    throw new Error(
      'saveSection: unknown payload key(s) for section "' + sectionNum + '": ' +
      unknownKeys.join(', ') +
      '. Allowed keys: ' + allowed.join(', ')
    );
  }

  // Light nested shape validation for child-row sections
  if (Object.prototype.hasOwnProperty.call(data, 'projects')) {
    validateArrayOfObjects(sectionNum, 'projects', data.projects);
  }

  if (Object.prototype.hasOwnProperty.call(data, 'forces')) {
    validateArrayOfObjects(sectionNum, 'forces', data.forces);
  }

  if (Object.prototype.hasOwnProperty.call(data, 'processes')) {
    validateArrayOfObjects(sectionNum, 'processes', data.processes);
  }
}

function mapPayloadToDbFields(data, fieldMap) {
  var output = {};
  var keys = Object.keys(fieldMap);

  for (var i = 0; i < keys.length; i++) {
    var payloadKey = keys[i];
    var dbField = fieldMap[payloadKey];

    if (Object.prototype.hasOwnProperty.call(data, payloadKey)) {
      output[dbField] = data[payloadKey];
    }
  }

  return output;
}

function applySectionScalarTargets(ss, sectionNum, interviewId, data) {
  var config = SECTION_FIELD_MAP[sectionNum];
  if (!config || !config.scalarTargets || !config.scalarTargets.length) return;

  for (var i = 0; i < config.scalarTargets.length; i++) {
    var target = config.scalarTargets[i];
    var dbFields = mapPayloadToDbFields(data, target.map);

    if (Object.keys(dbFields).length > 0) {
      updateInterviewFields(ss, target.sheet, interviewId, dbFields);
    }
  }
}

function validateTemplate(options) {
  var opts = options || {};
  var showSuccess = opts.showSuccess === true;

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetKeys = Object.keys(SHEETS);

  for (var i = 0; i < sheetKeys.length; i++) {
    var key = sheetKeys[i];
    var sheetName = SHEETS[key];

    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      throw new Error('Template error: missing sheet "' + sheetName + '".');
    }

    var expectedHeaders = HEADERS[sheetName];
    if (!expectedHeaders) continue; // CONFIG has no strict header contract

    var lastCol = sheet.getLastColumn();
    var actualHeaders = sheet.getRange(1, 1, 1, lastCol).getValues()[0];

    // --- STRICT COLUMN COUNT CHECK ---
    if (actualHeaders.length !== expectedHeaders.length) {
      throw new Error(
        'Template error in sheet "' + sheetName + '"\n\n' +
        'Expected ' + expectedHeaders.length + ' columns but found ' + actualHeaders.length + '.\n\n' +
        'Expected:\n' + expectedHeaders.join(', ') + '\n\n' +
        'Found:\n' + actualHeaders.join(', ')
      );
    }

    // --- COLUMN-BY-COLUMN VALIDATION ---
    for (var c = 0; c < expectedHeaders.length; c++) {
      var expected = expectedHeaders[c];
      var actual = String(actualHeaders[c]).trim();

      if (actual !== expected) {
        throw new Error(
          'Template error in sheet "' + sheetName + '"\n\n' +
          'Column #' + (c + 1) + ' mismatch\n\n' +
          'Expected: "' + expected + '"\n' +
          'Found: "' + actual + '"'
        );
      }
    }
  }

  if (showSuccess) {
    SpreadsheetApp.getUi().alert('✅ Template is valid.');
  }
}

function validateTemplateFromMenu() {
  validateTemplate({ showSuccess: true });
}

function repairTemplate() {
  var ui = SpreadsheetApp.getUi();

  try {
    validateTemplate();
    ui.alert('✅ Template already valid. No repair needed.');
    return;
  } catch (err) {
    var proceed = ui.alert(
      '⚠️ Template issue detected\n\n' +
      err.message + '\n\n' +
      'Do you want to RECREATE missing sheets only?\n\n' +
      'Note: incorrect existing headers will NOT be repaired automatically.\n' +
      'Use a fresh template or fix the header row manually.',
      ui.ButtonSet.YES_NO
    );

    if (proceed !== ui.Button.YES) return;
  }

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheetKeys = Object.keys(SHEETS);

  for (var i = 0; i < sheetKeys.length; i++) {
    var sheetName = SHEETS[sheetKeys[i]];
    var expectedHeaders = HEADERS[sheetName];

    // Skip CONFIG (no strict contract)
    if (!expectedHeaders) continue;

    var sheet = ss.getSheetByName(sheetName);

    // --- CREATE MISSING SHEET ---
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.getRange(1, 1, 1, expectedHeaders.length)
        .setValues([expectedHeaders])
        .setFontWeight('bold');

      sheet.setFrozenRows(1);
      continue;
    }

    // --- DO NOT MODIFY EXISTING STRUCTURE ---
    // (We only warn — no destructive changes)
  }

  ui.alert(
    '🛠 Repair complete\n\n' +
    '✔ Missing sheets recreated\n' +
    '⚠ Existing sheets with wrong headers were not modified.\n' +
    'For header mismatches, copy a fresh template or fix row 1 manually.'
  );
}

function ensureSetup() {
  try {
    validateTemplate({ showSuccess: false });
  } catch (err) {
    SpreadsheetApp.getUi().alert(
      'Template is corrupted or missing tabs.\n\n' +
      'Copy a fresh template.\n\n' +
      'Details:\n' + err.message
    );
    throw err;
  }
}

function buildInterviewId() {
  var timestamp = Utilities.formatDate(
    new Date(),
    Session.getScriptTimeZone(),
    'yyyyMMdd_HHmmss'
  );

  var chars = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  var suffix = '';

  for (var i = 0; i < 4; i++) {
    suffix += chars.charAt(Math.floor(Math.random() * chars.length));
  }

  return 'INT_' + timestamp + '_' + suffix;
}

// ==================== NEW / RESUME ====================

function newInterview() {
  ensureSetup();

  var id = buildInterviewId();

  var ss = SpreadsheetApp.getActiveSpreadsheet();

  // Init META
  // Write only columns A:E.
  // Column F ("status") is owned by the sheet/template formula and is NOT written by Apps Script.
  // The wizard must display that backend status as read-only information.
  var metaSheet = ss.getSheetByName(SHEETS.META);
  var nextMetaRow = metaSheet.getLastRow() + 1;
  metaSheet.getRange(nextMetaRow, 1, 1, 5).setValues([[
    id,
    Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd'),
    '',
    '',
    ''
  ]]);

  // Init SCOPE
  var scopeSheet = ss.getSheetByName(SHEETS.SCOPE);
  var scopeRow = buildEmptyRow(HEADERS.DB_SCOPE_VISION);
  setRowValue(scopeRow, HEADERS.DB_SCOPE_VISION, 'interview_id', id);
  scopeSheet.appendRow(scopeRow);

  SpreadsheetApp.flush();
  openWizard(id);
}

function resumeInterview() {
  ensureSetup();
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var activeSheet = ss.getActiveSheet();
  var activeRow = activeSheet.getActiveCell().getRow();
  var selectedId = String(activeSheet.getRange(activeRow, 1).getValue()).trim();

  if (selectedId.indexOf('INT_') === 0) {
    openWizard(selectedId);
    return;
  }

  var interviews = listInterviews();
  if (interviews.length === 0) { SpreadsheetApp.getUi().alert('No interview found.\n\n→ Use "▶ New interview" to create one.'); return; }
  if (interviews.length === 1) { openWizard(interviews[0].id); return; }

  SpreadsheetApp.getUi().alert(
    'To resume an interview:\n\n' +
    '1. Go to the "DASH_OVERVIEW" sheet.\n' +
    '2. Select the cell in column "Interview ID" for the interview you want to resume.\n' +
    '3. Click "↩ Resume interview" again.'
  );
}

function listInterviews() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SHEETS.META);
  var data = sheet.getDataRange().getValues();
  var headers = data[0];
  var result = [];
  for (var i = 1; i < data.length; i++) {
    var rid = String(data[i][0] || '').trim();
    if (rid && rid.indexOf('INT_') === 0) {
      result.push({
        id: rid,
        name: colVal(data[i], headers, 'interviewee_name') || '(unnamed)',
        role: colVal(data[i], headers, 'interviewee_role') || '',
        status: colVal(data[i], headers, 'status') || '?'
      });
    }
  }
  return result;
}

// Note: this reads the 1-to-1 META row and the full SCOPE/VISION row,
// which also contains S5, S7 and cross-section notes.
function loadInitialData(id) {
  if (!id) return null;
  var ss = SpreadsheetApp.getActiveSpreadsheet();

  var metaObj = getRowsFor(ss, SHEETS.META, id, 1)[0] || {};
  var scopeObj = getRowsFor(ss, SHEETS.SCOPE, id, 1)[0] || {};

  return {
    id: id,
    interview: Object.assign({}, metaObj, scopeObj)
  };
}

function loadSecondaryData(id) {

  return {
    projects: getRowsFor(ss, SHEETS.PROJECTS, id, 0),
    market: getRowsFor(ss, SHEETS.MARKET, id, 0),
    processes: getRowsFor(ss, SHEETS.PROCESSES, id, 0),
    execution: getRowsFor(ss, SHEETS.EXECUTION, id, 1)[0] || null
  };
}

function getRowsFor(ss, sheetName, interviewId, limit) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];

  // getDataRange() is the most performant way to read without knowing exact boundaries
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  var headers = getHeaders(sheet);
  var rows = [];
  var targetId = String(interviewId).trim();

  for (var i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() === targetId) {
      var obj = {};
      for (var j = 0; j < headers.length; j++) {
        var val = data[i][j];
        if (val instanceof Date) val = val.toISOString();
        obj[headers[j]] = val;
      }
      rows.push(obj);

      // OPTIMIZATION: Break early if we only needed a specific number of rows
      if (limit && rows.length >= limit) break;
    }
  }
  return rows;
}

// ==================== SAVE ====================

function saveSection(num, id, data) {
  if (!id) return false;

  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = data || {};
  validateSectionPayload(num, payload);

  switch (num) {
    case 1:
      applySectionScalarTargets(ss, 1, id, payload);
      break;

    case 2:
      if (Object.prototype.hasOwnProperty.call(payload, 'projects')) {
        replaceChildRows(ss, SHEETS.PROJECTS, id, (payload.projects || []).map(function (p) {
          return [id, p.name, p.phase, p.results, p.lessons];
        }));
      }
      applySectionScalarTargets(ss, 2, id, payload);
      break;

    case 3:
      if (Object.prototype.hasOwnProperty.call(payload, 'forces')) {
        replaceChildRows(ss, SHEETS.MARKET, id, FORCE_NAMES.map(function (fname, i) {
          var f = (payload.forces || [])[i] || {};
          return [id, fname, f.opportunities || '', f.scoreOpp || '', f.threats || '', f.scoreThr || ''];
        }));
      }
      applySectionScalarTargets(ss, 3, id, payload);
      break;

    case 4:
      if (Object.prototype.hasOwnProperty.call(payload, 'processes')) {
        replaceChildRows(ss, SHEETS.PROCESSES, id, (payload.processes || []).map(function (p) {
          return [
            id,
            p.name,
            p.accelerate,
            p.costs,
            p.quality,
            p.determinism,
            p.dataQuality,
            p.skillsBuyin,
            p.existingTools,
            p.comment
          ];
        }));
      }
      applySectionScalarTargets(ss, 4, id, payload);
      break;

    case 5:
      applySectionScalarTargets(ss, 5, id, payload);
      break;

    case 6:
      if (
        Object.prototype.hasOwnProperty.call(payload, 'actionPlan') ||
        Object.prototype.hasOwnProperty.call(payload, 'roadblocks') ||
        Object.prototype.hasOwnProperty.call(payload, 'velocity') ||
        Object.prototype.hasOwnProperty.call(payload, 'governance') ||
        Object.prototype.hasOwnProperty.call(payload, 'culture') ||
        Object.prototype.hasOwnProperty.call(payload, 'organization') ||
        Object.prototype.hasOwnProperty.call(payload, 'skills') ||
        Object.prototype.hasOwnProperty.call(payload, 'it')
      ) {
        replaceChildRows(ss, SHEETS.EXECUTION, id, [[
          id,
          payload.actionPlan,
          payload.roadblocks,
          payload.velocity,
          payload.governance,
          payload.culture,
          payload.organization,
          payload.skills,
          payload.it
        ]]);
      }
      applySectionScalarTargets(ss, 6, id, payload);
      break;

    case 7:
      applySectionScalarTargets(ss, 7, id, payload);
      break;

    default:
      throw new Error('saveSection: unknown section "' + num + '".');
  }

  return true;
}

function saveAllSections(id, payloadBySection) {
  if (!id) {
    throw new Error('saveAllSections: missing interview id.');
  }

  if (!isPlainObject(payloadBySection)) {
    throw new Error('saveAllSections: payloadBySection must be an object.');
  }

  // Save in strict section order so the finalization is deterministic
  for (var sectionNum = 1; sectionNum <= 7; sectionNum++) {
    var sectionPayload = payloadBySection[sectionNum] || payloadBySection[String(sectionNum)] || {};
    saveSection(sectionNum, id, sectionPayload);
  }

  return true;
}

// ==================== BATCH HELPERS (ULTRA PERFORMANCE) ====================

function findRowByInterviewId(sheet, interviewId) {
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return -1;

  var idValues = sheet.getRange(2, 1, lastRow - 1, 1).getValues();
  var targetId = String(interviewId).trim();

  for (var i = 0; i < idValues.length; i++) {
    if (String(idValues[i][0]).trim() === targetId) {
      return i + 2; // convert array index to sheet row number
    }
  }

  return -1;
}

function normalizeHeaderName(name) {
  return String(name == null ? '' : name).trim();
}

function getHeaderIndexMap(headers) {
  var map = {};
  for (var i = 0; i < headers.length; i++) {
    map[normalizeHeaderName(headers[i])] = i + 1; // 1-based column index
  }
  return map;
}

function updateInterviewFields(ss, sheetName, interviewId, fields) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) {
    throw new Error('updateInterviewFields: sheet "' + sheetName + '" not found.');
  }

  var fieldKeys = Object.keys(fields || {});
  if (fieldKeys.length === 0) {
    return;
  }

  var lastCol = sheet.getLastColumn();
  if (lastCol < 1) {
    throw new Error('updateInterviewFields: sheet "' + sheetName + '" has no headers.');
  }

  var headers = getHeaders(sheet);
  var headerMap = getHeaderIndexMap(headers);
  var rowNum = findRowByInterviewId(sheet, interviewId);

  if (rowNum === -1) {
    throw new Error(
      'updateInterviewFields: interview "' + interviewId + '" not found in sheet "' + sheetName + '". Data was NOT saved.'
    );
  }

  // Validate all requested fields before writing anything
  for (var i = 0; i < fieldKeys.length; i++) {
    var fieldName = fieldKeys[i];
    if (!headerMap[fieldName]) {
      throw new Error(
        'updateInterviewFields: field "' + fieldName + '" not found in sheet "' + sheetName + '".'
      );
    }
  }

  // Read full row once, patch in memory, write once
  var rowValues = sheet.getRange(rowNum, 1, 1, lastCol).getValues()[0];

  for (var j = 0; j < fieldKeys.length; j++) {
    var key = fieldKeys[j];
    var colIndex = headerMap[key] - 1; // convert 1-based column index to 0-based array index
    rowValues[colIndex] = fields[key];
  }

  sheet.getRange(rowNum, 1, 1, lastCol).setValues([rowValues]);
}

function replaceChildRows(ss, sheetName, interviewId, newRows) {
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(10000)) {
    throw new Error('replaceChildRows: could not obtain lock for sheet "' + sheetName + '". Please retry.');
  }

  try {
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      throw new Error('replaceChildRows: sheet "' + sheetName + '" not found.');
    }

    if (Object.prototype.toString.call(newRows) !== '[object Array]') {
      throw new Error('replaceChildRows: newRows must be an array for sheet "' + sheetName + '".');
    }

    var headers = getHeaders(sheet);
    var expectedWidth = headers.length;
    var targetId = String(interviewId).trim();
    var lastRow = sheet.getLastRow();

    // Validate every incoming row BEFORE writing anything
    for (var r = 0; r < newRows.length; r++) {
      var row = newRows[r];

      if (Object.prototype.toString.call(row) !== '[object Array]') {
        throw new Error(
          'replaceChildRows: row #' + (r + 1) + ' in sheet "' + sheetName + '" is not an array.'
        );
      }

      if (row.length !== expectedWidth) {
        throw new Error(
          'replaceChildRows: row #' + (r + 1) + ' in sheet "' + sheetName + '" has ' +
          row.length + ' column(s), expected ' + expectedWidth + '.'
        );
      }

      if (String(row[0]).trim() !== targetId) {
        throw new Error(
          'replaceChildRows: row #' + (r + 1) + ' in sheet "' + sheetName + '" has interview_id "' +
          row[0] + '" but expected "' + targetId + '".'
        );
      }
    }

    // Read existing rows below header
    var data = lastRow > 1
      ? sheet.getRange(2, 1, lastRow - 1, expectedWidth).getValues()
      : [];

    // Keep rows that belong to other interviews
    var keptRows = [];
    for (var i = 0; i < data.length; i++) {
      if (String(data[i][0]).trim() !== targetId) {
        keptRows.push(data[i]);
      }
    }

    // Append new rows for this interview
    for (var j = 0; j < newRows.length; j++) {
      keptRows.push(newRows[j]);
    }

    // Rewrite dataset below header
    if (keptRows.length > 0) {
      sheet.getRange(2, 1, keptRows.length, expectedWidth).setValues(keptRows);
    }

    // Clear leftover old rows if dataset shrank
    var expectedLastRow = keptRows.length + 1; // +1 header row
    if (lastRow > expectedLastRow) {
      sheet.getRange(expectedLastRow + 1, 1, lastRow - expectedLastRow, expectedWidth).clearContent();
    }
  } finally {
    lock.releaseLock();
  }
}

function getHeaders(sheet) {
  var raw = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];
  var headers = [];
  for (var i = 0; i < raw.length; i++) {
    headers.push(normalizeHeaderName(raw[i]));
  }
  return headers;
}

function buildEmptyRow(headers) { var r = []; for (var i = 0; i < headers.length; i++) r.push(''); return r; }
function setRowValue(row, headers, name, val) { var idx = headers.indexOf(name); if (idx >= 0) row[idx] = val; }

function colVal(row, headers, name) {
  var idx = headers.indexOf(normalizeHeaderName(name));
  return (idx >= 0) ? String(row[idx] || '') : '';
}

function hasText(value) {
  return !!String(value || '').trim();
}

function rowHasAnyContent(row, fields) {
  if (!row) return false;
  for (var i = 0; i < fields.length; i++) {
    if (hasText(row[fields[i]])) return true;
  }
  return false;
}

function arrayHasAnyContent(rows, fields) {
  if (!rows || !rows.length) return false;
  for (var i = 0; i < rows.length; i++) {
    if (rowHasAnyContent(rows[i], fields)) return true;
  }
  return false;
}

function recomputeSavedSectionsFromLoadedData() {
  var interview = loadedInitialInterview || {};
  var secondary = loadedSecondaryData || {};

  for (var i = 0; i <= 6; i++) {
    var rule = COMPLETION_RULES[i];
    var done = false;

    if (rule && rule.interviewFields && rowHasAnyContent(interview, rule.interviewFields)) {
      done = true;
    }

    if (!done && rule && rule.secondaryTable && rule.secondaryFields) {
      done = arrayHasAnyContent(secondary[rule.secondaryTable], rule.secondaryFields);
    }

    if (!done && rule && rule.secondaryRow && rule.secondaryFields) {
      done = rowHasAnyContent(secondary[rule.secondaryRow], rule.secondaryFields);
    }

    savedSections[i] = done;
  }

  updateNav();
}

  function flushQueuedSave() {
    if (!interviewId || isSaving) return;

    // If user changed something during the in-flight request,
    // ensure one more save is queued even if no explicit trigger fired again.
    if (queuedSaveSecIdx === null && isDirty) {
      queuedSaveSecIdx = current;
    }

    if (queuedSaveSecIdx === null) {
      if (pendingFinalize) {
        pendingFinalize = false;
        finalizeInterview();
      }
      return;
    }

    var nextSecIdx = queuedSaveSecIdx;
    queuedSaveSecIdx = null;
    saveInBackground(nextSecIdx);
  }

    // BACKGROUND SYNC ENGINE
    function saveInBackground(secIdx) {
    if (!interviewId) return;

    // One request at a time.
    // Keep only the latest requested section; the dirty flag ensures we retry.
    if (isSaving) {
      queuedSaveSecIdx = secIdx;
      isDirty = true;
      return;
    }

    var payload = SAVE_BUILDERS[secIdx + 1]();

    isSaving = true;
    isDirty = false;
    showStatus('saving', 'Synchronisation en arrière-plan...');

    google.script.run
      .withSuccessHandler(function() {
        isSaving = false;
        savedSections[secIdx] = true;
        updateNav();
        showStatus('save-ok', '✓ Données synchronisées');
        flushQueuedSave();
      })
      .withFailureHandler(function() {
        isSaving = false;
        isDirty = true;
        savedSections[secIdx] = false;
        updateNav();
        showStatus('save-err', '✗ En attente de réseau...');
        flushQueuedSave();
      })
      .saveSection(secIdx + 1, interviewId, payload);
  }
